<script>

// ============ RANKING ============

function getRanking() {

  return JSON.parse(localStorage.getItem("lampmanRanking")) || [];

}

 

function saveRanking(name, score) {

  let ranking = getRanking();

  ranking.push({ name, score });

  ranking.sort((a, b) => b.score - a.score);

  ranking = ranking.slice(0, 5);

  localStorage.setItem("lampmanRanking", JSON.stringify(ranking));

}

 

function showRanking() {

  const list = document.getElementById("ranking");

  list.innerHTML = "";

  getRanking().forEach(r => {

    const li = document.createElement("li");

    li.textContent = `${r.name} - ${r.score}`;

    list.appendChild(li);

  });

}

 

showRanking();

 

// ============ MENU ============

let gameStarted = false;

 

function startGame() {

  document.getElementById("menu").style.display = "none";

  document.getElementById("title").style.display = "none";

  canvas.style.display = "block";

  startMusic();

  gameStarted = true;

  loop();

}

 

// ============ AUDIO ============

let audioCtx;

let musicInterval;

 

function startMusic() {

  audioCtx = new (window.AudioContext || window.webkitAudioContext)();

  const notes = [262,330,392,330,294,330,440,392];

  let i = 0;

  musicInterval = setInterval(() => {

    const osc = audioCtx.createOscillator();

    const gain = audioCtx.createGain();

    osc.type = "square";

    osc.frequency.value = notes[i++ % notes.length];

    gain.gain.value = 0.04;

    osc.connect(gain);

    gain.connect(audioCtx.destination);

    osc.start();

    osc.stop(audioCtx.currentTime + 0.2);

  }, 250);

}

 

function playLoseLifeSound() {

  const osc = audioCtx.createOscillator();

  const gain = audioCtx.createGain();

  osc.type = "sawtooth";

  osc.frequency.setValueAtTime(400, audioCtx.currentTime);

  osc.frequency.exponentialRampToValueAtTime(100, audioCtx.currentTime + 0.4);

  gain.gain.value = 0.1;

  osc.connect(gain);

  gain.connect(audioCtx.destination);

  osc.start();

  osc.stop(audioCtx.currentTime + 0.4);

}

 

function playWinPhaseSound() {

  [523, 659, 784].forEach((n, i) => {

    const osc = audioCtx.createOscillator();

    const gain = audioCtx.createGain();

    osc.type = "square";

    osc.frequency.value = n;

    gain.gain.value = 0.06;

    osc.connect(gain);

    gain.connect(audioCtx.destination);

    osc.start(audioCtx.currentTime + i * 0.15);

    osc.stop(audioCtx.currentTime + i * 0.15 + 0.15);

  });

}

 

// ============ GAME ============

const canvas = document.getElementById("game");

const ctx = canvas.getContext("2d");

 

const tile = 20;

const rows = 20;

const cols = 20;

 

let phase = 1;

let score = 0;

let lives = 3;

 

let power = 0;

const maxPower = 600;

let frame = 0;

 

let lampMan;

let donkeyKong;

let map;

let donkeySpeed = 30;

 

function generateMaze() {

  let m = Array.from({ length: rows }, () =>

    Array.from({ length: cols }, () => 1)

  );

  for (let i = 0; i < phase * 5; i++) {

    m[Math.floor(Math.random()*rows)][Math.floor(Math.random()*cols)] = 0;

  }

  return m;

}

 

function placeCherries() {

  let c = 0;

  while (c < 4) {

    let x = Math.floor(Math.random()*cols);

    let y = Math.floor(Math.random()*rows);

    if (map[y][x] === 1) {

      map[y][x] = 2;

      c++;

    }

  }

}

 

function startPhase() {

  lampMan = { x: 1, y: 1 };

  donkeyKong = { x: cols-2, y: rows-2 };

  map = generateMaze();

  map[1][1] = 0;

  placeCherries();

  donkeySpeed = Math.max(5, 30 - phase * 2);

}

 

startPhase();

 

document.addEventListener("keydown", e => {

  if (!gameStarted) return;

 

  let nx = lampMan.x;

  let ny = lampMan.y;

 

  if (e.key === "ArrowUp") ny--;

  if (e.key === "ArrowDown") ny++;

  if (e.key === "ArrowLeft") nx--;

  if (e.key === "ArrowRight") nx++;

 

  if (nx>=0 && ny>=0 && nx<cols && ny<rows) {

    lampMan.x = nx;

    lampMan.y = ny;

    if (map[ny][nx] === 1) { map[ny][nx]=0; score+=10; }

    if (map[ny][nx] === 2) { map[ny][nx]=0; power=maxPower; score+=50; }

  }

});

 

function endGame(message) {

  let name = prompt(message + "\nDigite seu nome:");

  if (!name) name = "Jogador";

  saveRanking(name, score);

  location.reload();

}

 

function loseLife() {

  lives--;

  playLoseLifeSound();

  power = 0;

  lampMan.x = 1; lampMan.y = 1;

  donkeyKong.x = cols-2; donkeyKong.y = rows-2;

  if (lives <= 0) endGame("💀 GAME OVER");

}

 

function update() {

  frame++;

  if (frame % donkeySpeed === 0) {

    const d = [[1,0],[-1,0],[0,1],[0,-1]][Math.floor(Math.random()*4)];

    donkeyKong.x = Math.max(0,Math.min(cols-1,donkeyKong.x+d[0]));

    donkeyKong.y = Math.max(0,Math.min(rows-1,donkeyKong.y+d[1]));

  }

 

  if (power>0) power--;

 

  if (lampMan.x===donkeyKong.x && lampMan.y===donkeyKong.y) {

    if (power>0) {

      score+=500;

      donkeyKong.x=cols-2; donkeyKong.y=rows-2;

    } else loseLife();

  }

 

  if (!map.flat().some(v=>v===1||v===2)) {

    playWinPhaseSound();

    phase++;

    if (phase>10) endGame("🏆 VOCÊ ZEROU O JOGO!");

    startPhase();

  }

}

 

function draw() {

  ctx.clearRect(0,0,400,420);

 

  for (let y=0;y<rows;y++) for (let x=0;x<cols;x++) {

    if (map[y][x]===1 || map[y][x]===2) {

      ctx.fillStyle = map[y][x]===1 ? "white" : "red";

      ctx.beginPath();

      ctx.arc(x*tile+10,y*tile+10,map[y][x]===1?3:6,0,7);

      ctx.fill();

    }

  }

 

  ctx.fillStyle="yellow";

  ctx.beginPath();

  ctx.arc(lampMan.x*tile+10,lampMan.y*tile+10,8,0,7);

  ctx.fill();

 

  ctx.fillStyle=power>0?"blue":"brown";

  ctx.fillRect(donkeyKong.x*tile+4,donkeyKong.y*tile+4,12,12);

 

  ctx.fillStyle="white";

  ctx.fillText("Score: "+score,10,395);

  ctx.fillText("Fase: "+phase+"/10",300,395);

  ctx.fillText("Vidas: ❤️".repeat(lives),10,410);

 

  if (power>0) {

    ctx.fillStyle="red";

    ctx.fillRect(100,405,(power/maxPower)*200,8);

    ctx.strokeStyle="white";

    ctx.strokeRect(100,405,200,8);

  }

}

 

function loop() {

  if (!gameStarted) return;

  update();

  draw();

  requestAnimationFrame(loop);

}

    gain.gain.value = 0.06;

    osc.connect(gain);

    gain.connect(audioCtx.destination);

    osc.start(audioCtx.currentTime + i * 0.15);

    osc.stop(audioCtx.currentTime + i * 0.15 + 0.15);

  });

// ============ GAME ============

const canvas = document.getElementById("game");

const ctx = canvas.getContext("2d");

 

const tile = 20;

const rows = 20;

const cols = 20;

 

let phase = 1;

let score = 0;

let lives = 3;

 

let power = 0;

const maxPower = 600;

let frame = 0;

 

let lampMan;

let donkeyKong;

let map;

let donkeySpeed = 30;

 

function generateMaze() {

  let m = Array.from({ length: rows }, () =>

    Array.from({ length: cols }, () => 1)

  );

  for (let i = 0; i < phase * 5; i++) {

    m[Math.floor(Math.random()*rows)][Math.floor(Math.random()*cols)] = 0;

  }

  return m;

}

 

function placeCherries() {

  let c = 0;

  while (c < 4) {

    let x = Math.floor(Math.random()*cols);

    let y = Math.floor(Math.random()*rows);

    if (map[y][x] === 1) {

      map[y][x] = 2;

      c++;

    }

  }

}

 

function startPhase() {

  lampMan = { x: 1, y: 1 };

  donkeyKong = { x: cols-2, y: rows-2 };

  map = generateMaze();

  map[1][1] = 0;

  placeCherries();

  donkeySpeed = Math.max(5, 30 - phase * 2);

}

 

startPhase();

 

document.addEventListener("keydown", e => {

  if (!gameStarted) return;

 

  let nx = lampMan.x;

  let ny = lampMan.y;

 

  if (e.key === "ArrowUp") ny--;

  if (e.key === "ArrowDown") ny++;

  if (e.key === "ArrowLeft") nx--;

  if (e.key === "ArrowRight") nx++;

 

  if (nx>=0 && ny>=0 && nx<cols && ny<rows) {

    lampMan.x = nx;

    lampMan.y = ny;

    if (map[ny][nx] === 1) { map[ny][nx]=0; score+=10; }

    if (map[ny][nx] === 2) { map[ny][nx]=0; power=maxPower; score+=50; }

  }

});

 

function endGame(message) {

  let name = prompt(message + "\nDigite seu nome:");

  if (!name) name = "Jogador";

  saveRanking(name, score);

  location.reload();

}

 

function loseLife() {

  lives--;

  playLoseLifeSound();

  power = 0;

  lampMan.x = 1; lampMan.y = 1;

  donkeyKong.x = cols-2; donkeyKong.y = rows-2;

  if (lives <= 0) endGame("💀 GAME OVER");

}

 

function update() {

  frame++;

  if (frame % donkeySpeed === 0) {

    const d = [[1,0],[-1,0],[0,1],[0,-1]][Math.floor(Math.random()*4)];

    donkeyKong.x = Math.max(0,Math.min(cols-1,donkeyKong.x+d[0]));

    donkeyKong.y = Math.max(0,Math.min(rows-1,donkeyKong.y+d[1]));

  }

 

  if (power>0) power--;

 

  if (lampMan.x===donkeyKong.x && lampMan.y===donkeyKong.y) {

    if (power>0) {

      score+=500;

      donkeyKong.x=cols-2; donkeyKong.y=rows-2;

    } else loseLife();

  }

 

  if (!map.flat().some(v=>v===1||v===2)) {

    playWinPhaseSound();

    phase++;

    if (phase>10) endGame("🏆 VOCÊ ZEROU O JOGO!");

    startPhase();

  }

}

 

function draw() {

  ctx.clearRect(0,0,400,420);

 

  for (let y=0;y<rows;y++) for (let x=0;x<cols;x++) {

    if (map[y][x]===1 || map[y][x]===2) {

      ctx.fillStyle = map[y][x]===1 ? "white" : "red";

      ctx.beginPath();

      ctx.arc(x*tile+10,y*tile+10,map[y][x]===1?3:6,0,7);

      ctx.fill();

    }

  }

 

  ctx.fillStyle="yellow";

  ctx.beginPath();

  ctx.arc(lampMan.x*tile+10,lampMan.y*tile+10,8,0,7);

  ctx.fill();

 

  ctx.fillStyle=power>0?"blue":"brown";

  ctx.fillRect(donkeyKong.x*tile+4,donkeyKong.y*tile+4,12,12);

 

  ctx.fillStyle="white";

  ctx.fillText("Score: "+score,10,395);

  ctx.fillText("Fase: "+phase+"/10",300,395);

  ctx.fillText("Vidas: ❤️".repeat(lives),10,410);

 

  if (power>0) {

    ctx.fillStyle="red";

    ctx.fillRect(100,405,(power/maxPower)*200,8);

    ctx.strokeStyle="white";

    ctx.strokeRect(100,405,200,8);

  }

}

 

function loop() {

  if (!gameStarted) return;

  update();

  draw();

  requestAnimationFrame(loop);

}
